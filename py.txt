import customtkinter as ctk
import requests

# ----------------------------
# Configuration
# ----------------------------
API_KEY = "YOUR_API_KEY"

ctk.set_appearance_mode("dark")
ctk.set_default_color_theme("blue")

# ----------------------------
# Weather Function
# ----------------------------
def get_weather():
    city = city_entry.get()

    if city == "":
        result_label.configure(text="Please enter a city name")
        return

    url = f"https://api.openweathermap.org/data/2.5/weather?q={city}&appid={API_KEY}&units=metric"

    try:
        response = requests.get(url)
        data = response.json()

        if data["cod"] != 200:
            result_label.configure(text="City not found")
            return

        city_name = data["name"]
        temp = data["main"]["temp"]
        humidity = data["main"]["humidity"]
        wind = data["wind"]["speed"]
        weather = data["weather"][0]["description"].title()

        result = (
            f"📍 {city_name}\n\n"
            f"🌡 Temperature: {temp}°C\n"
            f"☁ Weather: {weather}\n"
            f"💧 Humidity: {humidity}%\n"
            f"🌬 Wind Speed: {wind} m/s"
        )

        result_label.configure(text=result)

    except Exception as e:
        result_label.configure(text=f"Error: {e}")


# ----------------------------
# Main Window
# ----------------------------
app = ctk.CTk()
app.title("Modern Weather App")
app.geometry("500x550")
app.resizable(False, False)

# Title
title_label = ctk.CTkLabel(
    app,
    text="🌤 Weather App",
    font=("Poppins", 28, "bold")
)
title_label.pack(pady=25)

# Search Box
city_entry = ctk.CTkEntry(
    app,
    width=300,
    height=40,
    placeholder_text="Enter City Name"
)
city_entry.pack(pady=10)

# Search Button
search_btn = ctk.CTkButton(
    app,
    text="Get Weather",
    width=150,
    height=40,
    command=get_weather
)
search_btn.pack(pady=15)

# Weather Display Card
weather_frame = ctk.CTkFrame(
    app,
    width=420,
    height=250,
    corner_radius=20
)
weather_frame.pack(pady=20)

result_label = ctk.CTkLabel(
    weather_frame,
    text="Search a city to view weather",
    font=("Arial", 18),
    justify="left"
)
result_label.place(relx=0.5, rely=0.5, anchor="center")

# Footer
footer = ctk.CTkLabel(
    app,
    text="Powered by OpenWeatherMap",
    text_color="gray"
)
footer.pack(side="bottom", pady=10)

app.mainloop()
